{%- comment -%} Lays out a list of cards in two columns: the first half on the left, the rest on the right. Entries with `hidden: true` are skipped. Params: items, plus `style` and `spacing` (passed through to card.html), col (column classes, default "col-lg-6 col-md-6 col-sm-12 mx-auto"). {%- endcomment -%} {%- assign visible_items = include.items | where_exp: "i", "i.hidden != true" -%} {%- assign half = visible_items.size | plus: 1 | divided_by: 2 -%} {%- assign col_class = include.col | default: "col-lg-6 col-md-6 col-sm-12 mx-auto" -%}
{%- for item in visible_items limit: half %} {% include card.html item=item style=include.style spacing=include.spacing %} {%- endfor %}
{%- for item in visible_items offset: half %} {% include card.html item=item style=include.style spacing=include.spacing %} {%- endfor %}
